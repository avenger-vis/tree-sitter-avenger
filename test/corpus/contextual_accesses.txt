  mark symbol as points {
    x: channel.x;
    adjust expr {
      x: item.channel.x;
      y: item.bbox.top;
    }
    derive text as labels {
      text: item.data."display label";
      x: item.channel.x;
    }
    view cartesian as viewport {
      x_probe: viewport.x.pixels;
      y_probe: viewport.y.domain.start;
    }
  }
  on pointermove as inspect {
    filter: datum."id" IS NOT NULL;
    set drag_x = event.coord.x;
    set canonicalized_x = EVENT.COORD.X;
    set start_x = event.start.coord.x;
    set domain_lo = event.domain.x.start;
    set domain_hi = event.domain.x.end;
    set drag_points = event.path;
    set facet_component = event.facet[1];
    set legend_hit = event.legend.value;
  }
}
